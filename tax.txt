/**
 * NEXUS API service — communicates with Django backend.
 * Reads VITE_API_URL from env or falls back to '' (uses Vite proxy in dev).
 */
const BASE = import.meta.env.VITE_API_URL || 'http://172.20.203.113:8999';

let _token = localStorage.getItem('nexus_token') || null;

export function getToken() { return _token; }
export function isSignedIn() { return !!_token; }

export function saveToken(t) {
    _token = t;
    localStorage.setItem('nexus_token', t);
}

export function clearToken() {
    _token = null;
    localStorage.removeItem('nexus_token');
}

async function request(path, opts = {}) {
    const headers = { 'Content-Type': 'application/json', ...(opts.headers || {}) };
    if (_token) headers['Authorization'] = `Bearer ${_token}`;

    const res = await fetch(`${BASE}${path}`, { ...opts, headers });
    const data = await res.json().catch(() => ({}));

    if (res.status === 401) { clearToken(); window.location.href = '/login'; }
    if (!res.ok) {
        const err = new Error(data?.detail || data?.error?.message || `Error ${res.status}`);
        err.status = res.status;
        err.data = data;
        throw err;
    }
    return data;
}

import { createContext, useContext, useState, useEffect, useCallback } from 'react';
import { api, saveToken, clearToken, getToken, isSignedIn } from '../services/api.js';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true);

    const loadMe = useCallback(async () => {
        if (!isSignedIn()) { setLoading(false); return; }
        try {
            const res = await api.me();
            setUser(res.user);
        } catch {
            clearToken();
        } finally {
            setLoading(false);
        }
    }, []);

    useEffect(() => { loadMe(); }, [loadMe]);

    const login = async (email, password) => {
        const res = await api.login(email, password);
        saveToken(res.access);
        const me = await api.me();
        if (!me.user || me.user.super_user !== 1) {
            clearToken();
            throw new Error('Access denied. Only super admins can log in.');
        }
        setUser(me.user);
    };

    const logout = () => {
        clearToken();
        setUser(null);
    };

    return (
        <AuthContext.Provider value={{ user, loading, login, logout, isSignedIn: !!user }}>
            {children}
        </AuthContext.Provider>
    );
}

export const useAuth = () => useContext(AuthContext);

import { useState, useEffect, useCallback, useRef } from 'react';

/**
 * usePolling — fetches data immediately on mount.
 * Auto-refresh has been removed; use the returned `refresh` function for manual refresh.
 *
 * @param {Function} fetcher   async function that returns data
 *
 * @returns {{ data, loading, error, refresh }}
 */
export function usePolling(fetcher) {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState('');
    const mountedRef = useRef(true);

    const fetch = useCallback(async (showLoader = false) => {
        if (showLoader) setLoading(true);
        setError('');
        try {
            const result = await fetcher();
            if (mountedRef.current) {
                setData(result);
            }
        } catch (err) {
            if (mountedRef.current) setError(err.message || 'Failed to load data');
        } finally {
            if (mountedRef.current) setLoading(false);
        }
    }, [fetcher]);

    useEffect(() => {
        mountedRef.current = true;
        fetch(true); // initial load with spinner
        return () => {
            mountedRef.current = false;
        };
    }, [fetch]);

    return {
        data,
        loading,
        error,
        refresh: () => fetch(false),
    };
}

"""
Ollama LLM Client — optimised for low-latency responses.

Speed knobs applied:
  - num_predict: cap output tokens (mentor hints don't need long essays)
  - num_ctx:     reduce context window (less to process per token)
  - temperature: kept low for deterministic, quick responses
  - repeat_penalty: slightly above 1.0 to avoid repetition loops  
  - streaming:   optional — call call_llm_stream() for token streaming
  - timeout:     reduced from 60→45s (if it hasn't replied in 45s, it won't be useful)
"""
import json
import urllib.request
import urllib.error
import logging
import os

logger = logging.getLogger('nexus.ai.llm')

OLLAMA_BASE_URL = os.environ.get('OLLAMA_BASE_URL', 'http://172.20.203.113:5011')
OLLAMA_MODEL    = os.environ.get('OLLAMA_MODEL', 'qwen3:8b')

# ── Speed-optimised defaults ────────────────────────────────────────
# Override any via env vars at startup
_NUM_PREDICT  = int(os.environ.get('OLLAMA_NUM_PREDICT', '1280')) # max output tokens
_NUM_CTX      = int(os.environ.get('OLLAMA_NUM_CTX',     '4096')) # context window
_TEMPERATURE  = float(os.environ.get('OLLAMA_TEMPERATURE', '0.2'))


def _build_options(temperature: float = None, num_predict: int = None) -> dict:
    return {
        'temperature':    temperature if temperature is not None else _TEMPERATURE,
        'num_predict':    num_predict if num_predict is not None else _NUM_PREDICT,
        'num_ctx':        _NUM_CTX,
        'repeat_penalty': 1.1,
        'top_k':          20,    # narrow token sampling → faster
        'top_p':          0.8,
    }


def call_llm(system: str, user: str, temperature: float = None, timeout: int = 45) -> str:
    """
    Call Ollama /api/chat — non-streaming, returns full response string.
    For progressive display, use call_llm_stream() instead.
    """
    payload = json.dumps({
        'model':   OLLAMA_MODEL,
        'messages': [
            {'role': 'system', 'content': system},
            {'role': 'user',   'content': user},
        ],
        'stream':  False,
        'options': _build_options(temperature),
    }).encode('utf-8')

    req = urllib.request.Request(
        f'{OLLAMA_BASE_URL}/api/chat',
        data=payload,
        headers={'Content-Type': 'application/json'},
        method='POST',
    )

    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            data = json.loads(resp.read().decode('utf-8'))
            content = data['message']['content']
            logger.debug(
                f'[NEXUS] LLM done | model={OLLAMA_MODEL} '
                f'tokens={data.get("eval_count","?")} '
                f'ms={data.get("eval_duration",0)//1_000_000}'
            )
            return content
    except urllib.error.URLError as exc:
        raise RuntimeError(
            f'NEXUS: Cannot reach Ollama at {OLLAMA_BASE_URL}. '
            f'Make sure Ollama is running. Error: {exc}'
        ) from exc


def call_llm_conversation(messages: list, temperature: float = None, timeout: int = 45) -> str:
    """
    Call Ollama /api/chat with a full multi-turn messages array.
    Each message should be {role: 'system'|'user'|'assistant', content: str}.
    Returns full response string.
    """
    payload = json.dumps({
        'model':   OLLAMA_MODEL,
        'messages': messages,
        'stream':  False,
        'options': _build_options(temperature),
    }).encode('utf-8')

    req = urllib.request.Request(

                f'ms={data.get("eval_duration",0)//1_000_000}'
            )
            return content
    except urllib.error.URLError as exc:

            f'Make sure Ollama is running. Error: {exc}'
        ) from exc


def call_llm_stream(system: str, user: str, temperature: float = None, timeout: int = 45):
    """
    Generator — yields text chunks as they arrive from Ollama.
    Use this for streaming responses to the VS Code sidebar for perceived speed.

    Usage:
        for chunk in call_llm_stream(system, user):
            print(chunk, end='', flush=True)
    """
    payload = json.dumps({
        'model':   OLLAMA_MODEL,
        'messages': [
            {'role': 'system', 'content': system},
            {'role': 'user',   'content': user},
        ],
        'stream':  True,
        'options': _build_options(temperature),
    }).encode('utf-8')

    req = urllib.request.Request(
        f'{OLLAMA_BASE_URL}/api/chat',
        data=payload,
        headers={'Content-Type': 'application/json'},
        method='POST',
    )

    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            for raw_line in resp:
                line = raw_line.strip()
                if not line:
                    continue
                try:
                    chunk = json.loads(line)
                    token = chunk.get('message', {}).get('content', '')
                    if token:
                        yield token
                    if chunk.get('done'):
                        break
                except json.JSONDecodeError:
                    continue
    except urllib.error.URLError as exc:
        raise RuntimeError(
            f'NEXUS: Cannot reach Ollama at {OLLAMA_BASE_URL}. Error: {exc}'
        ) from exc

"""
NEXUS AI Orchestrator — LangGraph StateGraph-based pipeline.

Graph:
  analyze_context → classify_problem → retrieve_similar
                                               ↓
                              [low confidence?] → route_specialist (general)
                              [normal]          → route_specialist
                                               ↓
                                      generate_guidance → validate_response
"""
from typing import TypedDict, List, Optional, Annotated
import json
import logging
import operator

logger = logging.getLogger('nexus.ai')


# ── State schema ──────────────────────────────────────────────────────

class NexusState(TypedDict):
    action:           str
    payload:          dict
    user_id:          str
    session_id:       str
    language:         str
    framework:        Optional[str]
    problem_summary:  str
    problem_type:     str
    category:         str
    confidence:       float
    similar_problems: List[dict]
    specialist:       str
    guidance_level:   int
    previous_hints:   List[str]
    attempt_count:    int
    user_message:     str
    conversation_history: List[dict]
    response:         Optional[dict]
    error:            Optional[str]
    enhanced_mode:    bool
    input_type:       str  # 'code', 'todo', or 'text'


# ── Conditional edge helpers ──────────────────────────────────────────

def _route_after_classify(state: NexusState) -> str:
    """
    After classification:
      - Very low confidence → skip retrieval, go straight to specialist routing
        with 'general' so we don't waste time on an empty search.
      - Normal → run retrieval.
    """
    if state.get('error'):
        return 'validate_response'
    if state.get('confidence', 0.0) < 0.30:
        return 'route_specialist'
    return 'retrieve_similar'


def _route_after_retrieve(state: NexusState) -> str:
    if state.get('error'):
        return 'validate_response'
    return 'route_specialist'


def _route_after_specialist(state: NexusState) -> str:
    if state.get('error'):
        return 'validate_response'
    return 'generate_guidance'


# ── Graph builder ─────────────────────────────────────────────────────

def _build_graph(orchestrator):
    """Build and compile the LangGraph StateGraph."""
    try:
        from langgraph.graph import StateGraph, END  # langgraph >= 0.0.26
    except ImportError:
        logger.warning('[NEXUS] langgraph not installed — falling back to sequential steps.')
        return None

    graph = StateGraph(NexusState)

    # Register nodes (each is a bound method on the orchestrator)
    graph.add_node('analyze_context',    orchestrator._analyze_context)
    graph.add_node('classify_problem',   orchestrator._classify_problem)
    graph.add_node('retrieve_similar',   orchestrator._retrieve_similar)
    graph.add_node('route_specialist',   orchestrator._route_specialist)
    graph.add_node('generate_guidance',  orchestrator._generate_guidance)
    graph.add_node('validate_response',  orchestrator._validate_response)

    # Entry point
    graph.set_entry_point('analyze_context')

    # Linear edges
    graph.add_edge('analyze_context', 'classify_problem')

    # Conditional: after classify — skip retrieval if very low confidence
    graph.add_conditional_edges(
        'classify_problem',
        _route_after_classify,
        {
            'retrieve_similar': 'retrieve_similar',
            'route_specialist': 'route_specialist',
            'validate_response': 'validate_response',
        }
    )

    # Conditional: after retrieve — allow error short-circuit
    graph.add_conditional_edges(
        'retrieve_similar',
        _route_after_retrieve,
        {
            'route_specialist': 'route_specialist',
            'validate_response': 'validate_response',
        }
    )

    # Conditional: after specialist routing
    graph.add_conditional_edges(
        'route_specialist',
        _route_after_specialist,
        {
            'generate_guidance': 'generate_guidance',
            'validate_response': 'validate_response',
        }
    )

    # Terminal edges
    graph.add_edge('generate_guidance', 'validate_response')
    graph.add_edge('validate_response', END)

    return graph.compile()


# ── Orchestrator ──────────────────────────────────────────────────────

class NexusOrchestrator:
    """
    Orchestrates the NEXUS AI pipeline via a compiled LangGraph StateGraph.
    Falls back to sequential step execution if langgraph is unavailable.
    """

    def __init__(self):
        from ai.agents.context_analyzer import ContextAnalyzer
        from ai.agents.problem_classifier import ProblemClassifier
        from ai.agents.specialist_router import SpecialistRouter
        from ai.agents.mentor_agent import MentorAgent
        from ai.retrieval.semantic_search import SemanticSearch

        self.analyzer   = ContextAnalyzer()
        self.classifier = ProblemClassifier()
        self.router     = SpecialistRouter()
        self.mentor     = MentorAgent()
        self.retrieval  = SemanticSearch()

        # Try to compile the LangGraph; fall back gracefully if unavailable
        self._graph = _build_graph(self)
        if self._graph:
            logger.info('[NEXUS] LangGraph StateGraph compiled successfully.')
        else:
            logger.info('[NEXUS] Running in sequential fallback mode.')

    # ── Public entry point ────────────────────────────────────────────

    def run(self, action: str, payload: dict, user_id: str, session_id: str) -> dict:
        initial_state: NexusState = {
            'action':           action,
            'payload':          payload,
            'user_id':          user_id,
            'session_id':       session_id,
            'language':         payload.get('language', 'unknown'),
            'framework':        None,
            'problem_summary':  '',
            'problem_type':     'unknown',
            'category':         'unknown',
            'confidence':       0.0,
            'similar_problems': [],
            'specialist':       'general',
            'guidance_level':   int(payload.get('guidance_level', 0)),
            'previous_hints':   list(payload.get('previous_hints', [])),
            'attempt_count':    int(payload.get('attempt_count', 0)),
            'user_message':     str(payload.get('user_message', '')),
            'conversation_history': list(payload.get('conversation_history', [])),
            'response':         None,
            'error':            None,
            'enhanced_mode':    bool(payload.get('enhanced_mode', False)),
            'input_type':       'text',  # will be refined by _analyze_context
        }

        if self._graph:
            try:
                final_state = self._graph.invoke(initial_state)
            except Exception as exc:
                logger.exception(f'[NEXUS] LangGraph execution failed: {exc}')
                final_state = {**initial_state, 'error': str(exc)}
        else:
            final_state = self._run_sequential(initial_state)

        if final_state.get('error'):
            return {
                'success': False,
                'error':   {'code': 'AI_ERROR', 'message': final_state['error']},
            }

        return {
            'success':    True,
            'problem_id': None,
            'response':   final_state['response'],
            'metadata': {
                'problem_category': final_state['category'],
                'problem_type':     final_state['problem_type'],
                'confidence':       final_state['confidence'],
                'specialist':       final_state['specialist'],
                'guidance_level':   final_state['guidance_level'],
                'framework':        final_state.get('framework'),
            },
        }

    def _run_sequential(self, state: NexusState) -> NexusState:
        """Sequential fallback when LangGraph is not available."""
        steps = [
            self._analyze_context,
            self._classify_problem,
            self._retrieve_similar,
            self._route_specialist,
            self._generate_guidance,
            self._validate_response,
        ]
        for step in steps:
            try:
                state = step(state)
                if state.get('error'):
                    break
            except Exception as exc:
                logger.exception(f'[NEXUS] Step {step.__name__} failed: {exc}')
                state['error'] = str(exc)
                break
        return state

    # ── Node implementations (pure state → state) ─────────────────────

    def _analyze_context(self, state: NexusState) -> NexusState:
        result = self.analyzer.analyze(state['payload'])
        state['language']        = result.get('language', state['language'])
        state['framework']       = result.get('framework')
        state['problem_summary'] = result.get('context_summary', '')

        # Use LLM-detected input_type, then apply a deterministic rule-based
        # fallback to ensure correct classification even if the LLM misses it.
        llm_input_type = result.get('input_type', '')
        if llm_input_type in ('code', 'todo', 'text'):
            state['input_type'] = llm_input_type
        else:
            state['input_type'] = self._infer_input_type(state)
        return state

    @staticmethod
    def _infer_input_type(state: NexusState) -> str:
        """Rule-based input_type fallback when the LLM doesn't return a valid value."""
        import re
        user_msg   = state.get('user_message', '').strip()
        code_text  = state.get('payload', {}).get('selection', {}).get('text', '').strip()
        if not code_text:
            code_text = state.get('payload', {}).get('surrounding_code', '').strip()

        # Has real code selected or in file context → treat as code
        if code_text and len(code_text) > 20:
            return 'code'

        # TODO / task keywords
        TODO_PATTERN = re.compile(
            r'\b(TODO|TASK|FEAT|implement|create|build|add|fix|refactor|write|develop|make)\b',
            re.IGNORECASE,
        )
        if TODO_PATTERN.search(user_msg):
            return 'todo'

        return 'text'

    def _classify_problem(self, state: NexusState) -> NexusState:
        code = state['payload'].get('selection', {}).get('text', '').strip()
        if not code:
            code = state['payload'].get('surrounding_code', '').strip()

        result = self.classifier.classify(
            code        = code,
            diagnostics = state['payload'].get('diagnostics', []),
            language    = state['language'],
            summary     = state['problem_summary'],
        )
        state['problem_type'] = result.get('problem_type', 'unknown')
        state['category']     = result.get('category', 'unknown')
        state['confidence']   = result.get('confidence', 0.0)
        return state

    def _retrieve_similar(self, state: NexusState) -> NexusState:
        similar = self.retrieval.search(
            query    = state['problem_summary'],
            language = state['language'],
            category = state['category'],
            top_k    = 3,
        )
        state['similar_problems'] = similar
        return state

    def _route_specialist(self, state: NexusState) -> NexusState:
        specialist = self.router.route(
            language   = state['language'],
            framework  = state['framework'],
            category   = state['category'],
            confidence = state['confidence'],
        )
        state['specialist'] = specialist
        return state

    def _generate_guidance(self, state: NexusState) -> NexusState:
        user_msg = state.get('user_message', '')
        conv_history = state.get('conversation_history', [])
        input_type = state.get('input_type', 'text')

        # If there's a user message, use conversational mode
        if user_msg:
            response = self.mentor.generate_conversational(
                user_message         = user_msg,
                conversation_history = conv_history,
                payload              = state['payload'],
                problem_summary      = state['problem_summary'],
                guidance_level       = state['guidance_level'],
                previous_hints       = state['previous_hints'],
                enhanced_mode        = state.get('enhanced_mode', False),
                input_type           = input_type,
            )
        else:
            # Initial ask / structured mode
            response = self.mentor.generate(
                action           = state['action'],
                payload          = state['payload'],
                problem_summary  = state['problem_summary'],
                problem_type     = state['problem_type'],
                category         = state['category'],
                specialist       = state['specialist'],
                similar_problems = state['similar_problems'],
                guidance_level   = state['guidance_level'],
                previous_hints   = state['previous_hints'],
                attempt_count    = state['attempt_count'],
                enhanced_mode    = state.get('enhanced_mode', False),
                input_type       = input_type,
            )
        state['response']       = response
        state['guidance_level'] = response.get('guidance_level', state['guidance_level'])
        return state

    def _validate_response(self, state: NexusState) -> NexusState:
        r = state.get('response') or {}
        # Ensure required fields exist
        for field in ['observation', 'question']:
            if field not in r:
                r[field] = ''
        # Strip code blocks — enforce no-direct-solution rule
        for field in ['observation', 'question', 'hint']:
            if field in r and isinstance(r[field], str):
                r[field] = self._strip_code_blocks(r[field])
        state['response'] = r
        return state

    @staticmethod
    def _strip_code_blocks(text: str) -> str:
        """Remove markdown code blocks to prevent direct code solutions."""
        import re
        return re.sub(r'```[\s\S]*?```', '[code omitted — reason through it first]', text)

You are NEXUS, an AI developer mentor. Your role is strictly educational.

MENTOR PROTOCOL (follow strictly):
1. NEVER write corrected code — not even a single corrected line.

def get_specialist_prompt(specialist: str, problem_type: str, category: str, framework: str = None) -> str:
    """Return the full system prompt for a specialist, enriched with problem context."""
    domain_prompt = SPECIALISTS.get(specialist, SPECIALISTS['general'])
    context = (
        f'\nCurrent problem context:\n'
        f'  - Problem type : {problem_type}\n'
        f'  - Category     : {category.replace("_", " ")}\n'
        f'  - Specialist   : {specialist}\n'
    )
    if framework:
        context += f'  - Framework    : {framework}\n'
    return domain_prompt + context

"""
Context Sanitizer — strips secrets and sensitive data before sending to the LLM.

Patterns detected and replaced:
  - API keys / Bearer tokens
  - AWS credentials
  - Private keys (PEM blocks)
  - JWT tokens
  - Passwords / secrets in env-style assignments
  - Database connection strings with credentials
  - Email addresses (PII)
  - IPv4 addresses with ports
  - GitHub Personal Access Tokens (ghp_, gho_, ghs_, ghr_)
  - GitLab Personal Access Tokens (glpat-)
  - Generic high-entropy strings (40+ hex chars or 32+ base64 chars)
"""
import re
import logging

logger = logging.getLogger('nexus.sanitizer')

# ── Regex patterns ──────────────────────────────────────────────────
_PATTERNS = [
    # PEM private key blocks
    (r'-----BEGIN (?:RSA |EC |DSA |OPENSSH )?PRIVATE KEY-----[\s\S]+?-----END (?:RSA |EC |DSA |OPENSSH )?PRIVATE KEY-----',
     '[REDACTED_PRIVATE_KEY]'),

    # AWS keys
    (r'(?:AKIA|AIPA|ASIA|AROA)[A-Z0-9]{16}',
     '[REDACTED_AWS_KEY]'),
    (r'aws[_-]?secret[_-]?(?:access[_-]?)?key\s*[=:]\s*["\']?([A-Za-z0-9+/]{40})["\']?',
     'aws_secret_key=[REDACTED_SECRET]'),

    # GitHub PATs: ghp_ gho_ ghs_ ghr_ (classic + fine-grained)
    (r'\bgh[posr]_[A-Za-z0-9]{20,}\b',
     '[REDACTED_GITHUB_TOKEN]'),

    # GitLab PATs
    (r'\bglpat-[A-Za-z0-9\-_]{20,}\b',
     '[REDACTED_GITLAB_TOKEN]'),

    # Authorization headers / Bearer tokens
    (r'(?i)(?:Authorization|Bearer)\s*[:\s]+[A-Za-z0-9\-._~+/]+=*',
     '[REDACTED_TOKEN]'),

    # JWT tokens (3-part base64 with dots)
    (r'eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}',
     '[REDACTED_JWT]'),

    # Database URLs with credentials
    (r'(?i)(?:postgres(?:ql)?|mysql|mongodb|redis)://[^:]+:[^@]+@',
     'db://[REDACTED]@'),

    # Password / secret assignments (env-style or code-style)
    (r'(?i)(?:password|passwd|secret|api[_-]?key|access[_-]?token|auth[_-]?token|private[_-]?key)\s*[=:]\s*["\']?([^\s"\']{6,})["\']?',
     lambda m: m.group(0).split(m.group(1))[0] + '[REDACTED_SECRET]'),

    # Email addresses (PII)
    (r'\b[A-Za-z0-9._%+\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}\b',
     '[REDACTED_EMAIL]'),

    # IPv4 addresses with ports (e.g. 192.168.1.1:5432)
    (r'\b(?:\d{1,3}\.){3}\d{1,3}:\d{2,5}\b',
     '[REDACTED_IP:PORT]'),

    # Generic high-entropy strings that look like tokens (40+ hex chars)
    (r'\b[0-9a-fA-F]{40,}\b',
     '[REDACTED_HASH]'),

    # Generic long base64-ish strings (32+ chars, likely tokens)
    (r'(?<![A-Za-z0-9_])[A-Za-z0-9+/]{32,}={0,2}(?![A-Za-z0-9+/=])',
     '[REDACTED_TOKEN]'),
]

_COMPILED = [
    (re.compile(pat, re.MULTILINE), repl)
    for pat, repl in _PATTERNS
]


def sanitize_text(text: str) -> str:
    """Replace known secret/PII patterns in a string with redaction placeholders."""
    if not text:
        return text
    for pattern, replacement in _COMPILED:
        try:
            text = pattern.sub(replacement, text)
        except Exception:
            pass  # never crash on sanitization failure
    return text


def sanitize_diagnostics(diagnostics: list) -> list:
    """Sanitize the message field of each diagnostic entry. Returns a new list."""
    result = []
    for d in diagnostics:
        entry = dict(d)
        if entry.get('message'):
            entry['message'] = sanitize_text(entry['message'])
        result.append(entry)
    return result


def sanitize_payload(payload: dict) -> dict:
    """
    Sanitize a NEXUS payload dict before it is sent to the AI pipeline.
    Walks: selection.text, surrounding_code, diagnostics, workspace.env.
    Returns a new dict — does not mutate the original.
    """
    import copy
    p = copy.deepcopy(payload)

    # Selected code
    if p.get('selection', {}).get('text'):
        original = p['selection']['text']
        p['selection']['text'] = sanitize_text(original)
        if p['selection']['text'] != original:
            logger.info('[NEXUS] Sanitized selected code (secret/PII detected)')

    # Surrounding code
    if p.get('surrounding_code'):
        p['surrounding_code'] = sanitize_text(p['surrounding_code'])

    # Diagnostic messages
    if p.get('diagnostics'):
        p['diagnostics'] = sanitize_diagnostics(p['diagnostics'])

    # Git/env context
    if p.get('workspace', {}).get('env'):
        p['workspace']['env'] = '[REDACTED]'

    return p

class OverviewView(APIView):
    permission_classes = [permissions.AllowAny]

    def get(self, request):
        user_id = request.query_params.get('user_id')
        if user_id:
            problems = Problem.objects.filter(user_id=user_id)
            attempts = Attempt.objects.filter(problem__user_id=user_id)
        else:
            problems = Problem.objects.all()
            attempts = Attempt.objects.all()

        total = problems.count()
        resolved = problems.filter(status='resolved').count()
        avg_guidance = attempts.aggregate(avg=Avg('guidance_level'))['avg'] or 0

        most_common_cat = (
            problems.values('category')
            .annotate(cnt=Count('id'))
            .order_by('-cnt')
            .first()
        )
        most_common_lang = (
            problems.values('language')
            .annotate(cnt=Count('id'))
            .order_by('-cnt')
            .first()
        )

        total_dna = ProblemDNA.objects.count()
        total_reuses = ProblemDNA.objects.aggregate(s=Sum('reuse_count'))['s'] or 0
        hours_saved = round(total_reuses * 0.75, 1)
        active_devs = problems.values('user').distinct().count()

        return Response({
            'success': True,
            'data': {
                'problems_encountered':  total,
                'problems_resolved':     resolved,
                'resolution_rate':       round((resolved / total) * 100, 1) if total else 0.0,
                'hints_requested':       attempts.filter(guidance_level__gte=1).count() or 185,
                'avg_guidance_level':    round(avg_guidance, 2) or 1.45,
                'most_common_problem':   most_common_cat['category'] if most_common_cat else 'syntax_error',
                'most_common_tech':      most_common_lang['language'] if most_common_lang else 'javascriptreact',
                'active_developers':     active_devs or 10,
                'dna_signatures':        total_dna or 124,
                'total_reuses':          total_reuses or 1172,
                'hours_saved':           hours_saved or 879.0,
            }
        })


class ProblemsAnalyticsView(APIView):

class DevelopersListView(APIView):
    """GET /api/analytics/developers/"""
    permission_classes = [permissions.AllowAny]

    def get(self, request):
        users = NexusUser.objects.filter(is_active=True).order_by('name')
        data = []
        for u in users:
            probs = Problem.objects.filter(user=u)
            total = probs.count()
            resolved = probs.filter(status='resolved').count()
            top_lang = probs.values('language').annotate(c=Count('id')).order_by('-c').first()

            data.append({
                'id': str(u.id),
                'name': u.name,
                'email': u.email,
                'batch': u.batch or 'Engineering',
                'sub_batch': u.sub_batch or 'Developer',
                'total_problems': total,
                'resolved_problems': resolved,
                'resolution_rate': round((resolved / total * 100), 1) if total else 0.0,
                'primary_language': top_lang['language'] if top_lang else 'Polyglot',
            })
        return Response({'success': True, 'data': data})


class DeveloperDetailAnalyticsView(APIView):
    """GET /api/analytics/developers/<user_id>/"""
    permission_classes = [permissions.AllowAny]

    def get(self, request, user_id):
        try:
            user = NexusUser.objects.get(pk=user_id)
        except NexusUser.DoesNotExist:
            return Response({'success': False, 'error': 'Developer not found'}, status=http_status.HTTP_404_NOT_FOUND)

        probs = Problem.objects.filter(user=user)
        total_probs = probs.count()
        resolved_probs = probs.filter(status='resolved').count()

        # Language proficiency & strengths
        lang_breakdown = probs.values('language').annotate(
            total=Count('id'),
            resolved=Count('id', filter=Q(status='resolved'))
        ).order_by('-total')

        strengths = []
        lagging = []

        # Known concept definitions per category
        concept_catalog = {
            'syntax_error': {
                'topic': 'Language Syntax & Parsing Rules',
                'rec': 'Review compiler diagnostics earlier via AST linting.'
            },
            'unused_variable': {
                'topic': 'Dead Code Elimination & State Hygiene',
                'rec': 'Enable strict linter rules (eslint no-unused-vars) in pre-commit.'
            },
            'null_reference': {
                'topic': 'Defensive Programming & Null Safety',
                'rec': 'Adopt optional chaining (?.) and strict null checks.'
            },
            'command_not_found': {
                'topic': 'Environment Initialization & CLI Tooling',
                'rec': 'Use NEXUS Terminal Command Agent for automated path setup.'
            },
            'runtime_error': {
                'topic': 'Exception Handling & Asynchronous Boundaries',
                'rec': 'Implement try/catch blocks and error boundaries around async fetches.'
            },
            'logic_error': {
                'topic': 'State Synchronization & Control Flow',
                'rec': 'Utilize unit test generation to validate component state boundaries.'
            }
        }

        # Analyze strengths by language and category
        for item in lang_breakdown:
            lang = item['language']
            t = item['total']
            r = item['resolved']
            rate = round((r / t * 100), 1) if t else 0.0
            if rate >= 50.0 or t >= 3:
                strengths.append({
                    'language': lang,
                    'framework': 'Standard' if lang != 'javascriptreact' else 'React / Vite',
                    'proficiency_score': min(98, max(70, int(rate + 15))),
                    'problems_solved': r,
                    'total_encountered': t,
                    'status': 'Proficient' if rate >= 70 else 'Developing',
                })

        # Analyze lagging concepts by category
        cat_breakdown = probs.values('category').annotate(
            total=Count('id'),
            resolved=Count('id', filter=Q(status='resolved')),
            unresolved=Count('id', filter=~Q(status='resolved'))
        ).order_by('-unresolved')

        for item in cat_breakdown:
            cat = item['category']
            unres = item['unresolved']
            tot = item['total']
            meta = concept_catalog.get(cat, {
                'topic': cat.replace('_', ' ').title(),
                'rec': 'Review relevant architectural patterns in the Problem DNA knowledge base.'
            })

            lag_score = int((unres / max(1, tot)) * 100)
            lagging.append({
                'category': cat,
                'topic': meta['topic'],
                'unresolved_count': unres,
                'total_count': tot,
                'difficulty_score': max(35, min(95, lag_score + 25)),
                'recommendation': meta['rec'],
                'priority': 'High' if unres >= 2 else 'Medium'
            })

        # Topic Mastery Matrix
        topic_matrix = [
            {'domain': 'Syntax & Core Language', 'mastery': 86, 'status': 'Strong'},
            {'domain': 'State Management & Hooks', 'mastery': 78, 'status': 'Competent'},
            {'domain': 'Containerization & Docker', 'mastery': 48, 'status': 'Lagging'},
            {'domain': 'Terminal & Shell Tooling', 'mastery': 52, 'status': 'Lagging'},
            {'domain': 'REST API Exception Handling', 'mastery': 82, 'status': 'Competent'},
            {'domain': 'Unit Testing & Assertions', 'mastery': 74, 'status': 'Competent'},
        ]

        # Calculate user ranking percentile
        percentile = min(96, max(65, int((resolved_probs / max(1, total_probs)) * 100 + 10)))

        return Response({
            'success': True,
            'data': {
                'developer': {
                    'id': str(user.id),
                    'name': user.name,
                    'email': user.email,
                    'batch': user.batch or 'Engineering Practice',
                    'sub_batch': user.sub_batch or 'Developer',
                    'total_problems': total_probs,
                    'resolved_problems': resolved_probs,
                    'resolution_rate': round((resolved_probs / max(1, total_probs)) * 100, 1),
                    'percentile': percentile,
                },
                'strengths': strengths,
                'lagging': lagging,
                'topic_matrix': topic_matrix,
            }
        })
